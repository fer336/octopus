Pegar este código en el módulo de la hoja "Cotizador" (no en módulo estándar):

1) ALT+F11
2) En el panel izquierdo: Microsoft Excel Objects > Hoja "Cotizador"
3) Pegar:

------------------------------------------------------------
Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo SafeExit
    If Application.EnableEvents = False Then Exit Sub

    Application.EnableEvents = False

    ' Si cambia código de producto en B8:B31, autocompleta
    If Not Intersect(Target, Me.Range("B8:B31")) Is Nothing Then
        Dim c As Range
        For Each c In Intersect(Target, Me.Range("B8:B31"))
            If c.Row >= 8 And c.Row <= 31 Then
                Call CompletarProductoPorCodigo(c.Row)
            End If
        Next c
    End If

    ' Renumerar items cuando cambien códigos
    If Not Intersect(Target, Me.Range("B8:B31")) Is Nothing Then
        Call RenumerarItems
    End If

SafeExit:
    Application.EnableEvents = True
End Sub
------------------------------------------------------------

Private Sub Workbook_Open()
